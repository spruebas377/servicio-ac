-- 1. Crear tabla de comentarios
create table if not exists public.profile_comments (
  id uuid not null default gen_random_uuid(),
  profile_id uuid not null,
  author_id uuid not null default auth.uid(),
  content text not null,
  created_at timestamp with time zone not null default now(),
  constraint profile_comments_pkey primary key (id),
  constraint profile_comments_profile_id_fkey foreign key (profile_id) references auth.users (id) on delete cascade,
  constraint profile_comments_author_id_fkey foreign key (author_id) references auth.users (id) on delete cascade,
  constraint profile_comments_no_self_comment check (profile_id <> author_id),
  constraint profile_comments_content_check check (char_length(trim(both from content)) > 0)
) tablespace pg_default;

-- 2. Índice para consultas rápidas por perfil
create index if not exists profile_comments_profile_created_idx 
  on public.profile_comments using btree (profile_id, created_at desc) tablespace pg_default;

-- 3. Habilitar Row Level Security (RLS)
alter table public.profile_comments enable row level security;

-- 4. Lectura pública (cualquiera puede leer comentarios)
create policy "profile_comments_select_public"
  on public.profile_comments
  for select
  using (true);

-- 5. Creación solo para usuarios autenticados en perfiles ajenos
create policy "profile_comments_insert_authenticated"
  on public.profile_comments
  for insert
  with check (
    auth.uid() = author_id and
    auth.uid() <> profile_id
  );

-- 6. Eliminación permitida al autor del comentario o al dueño del perfil
create policy "profile_comments_delete_author_or_owner"
  on public.profile_comments
  for delete
  using (
    auth.uid() = author_id or
    auth.uid() = profile_id
  );

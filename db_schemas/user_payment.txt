create table public.user_payment (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  id_user uuid null,
  id_payment_method bigint null,
  constraint user_payment_pkey primary key (id),
  constraint user_payment_id_payment_method_fkey foreign KEY (id_payment_method) references payment_method (id),
  constraint user_payment_id_user_fkey foreign KEY (id_user) references "user-data" (id)
) TABLESPACE pg_default;

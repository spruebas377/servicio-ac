create table public.user-data (
  id uuid not null,
  created_at timestamp with time zone not null default now(),
  name text null,
  phone text null,
  auth_id uuid null default auth.uid (),
  email text null,
  city bigint null,
  province bigint null,
  description text null default 'Usuario'::text,
  verified boolean null default false,
  avatar_url text null,
  age smallint null,
  height real null,
  hair text null,
  eyes text null,
  payment_method_id bigint[] null,
  gender bigint null,
  is_public boolean null default true,
  nationality text null,
  constraint user - data_pkey primary key (id),
  constraint user - data_city_fkey foreign KEY (city) references localidades (id) on update CASCADE on delete CASCADE,
  constraint user - data_gender_fkey foreign KEY (gender) references gender (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;

create table public.user_places (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  user_id uuid null default gen_random_uuid (),
  place_id bigint null,
  constraint user_place_pkey primary key (id),
  constraint user_place_place_id_fkey foreign KEY (place_id) references places (id),
  constraint user_place_user_id_fkey foreign KEY (user_id) references "user-data" (id)
) TABLESPACE pg_default;
